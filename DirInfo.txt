./                          # 项目根目录
├── web/                    # 前端应用（核心）
│   ├── index.html          # 主页面
│   ├── css/styles.css      # 样式
│   ├── js/                 # JS 脚本，本项目核心逻辑
│   │   ├── app.js          # 入口与路由
│   │   ├── chart.js        # 综合曲线图渲染
│   │   ├── store.js        # 数据层（localStorage）
│   │   ├── meds.js         # 药品库模块
│   │   ├── records.js      # 记录模块
│   │   ├── settings.js     # 设置模块
│   │   ├── theme.js        # 主题系统
│   │   ├── platform.js     # 平台检测与桥接
│   │   └── i18n.js         # 国际化引擎
│   ├── locales/            # 语言文件
│   │   ├── zh-CN.json      # 简体中文
│   │   └── en-US.json      # 美式英语
│   └── MedDB.json          # 内置药品数据库
├── desktop/                # 桌面端（Python + pywebview）
│   ├── main.py             # 桌面端入口
│   └── requirements.txt    # Python 依赖
├── android/                # Android 端（Kotlin）
│   ├── android/app/build.gradle.kts # 在这里修改移动端版本号
│   ├── app/src/main/assets/web # 这是到./web的符号链接，两个文件夹内容自动同步
│   └── app/src/main/java/org/jimmy/bdhub/MainActivity.kt # 主活动
├── assets/                 # 资产（图标等）
├── Build.sh                # Linux AppImage 构建脚本
├── Build.bat               # Windows exe 构建脚本
├── Build.spec              # PyInstaller 配置文件
├── AppImageBuilder.yml     # AppImage Builder 配置，在这里修改桌面端版本号
├── Stear.sh                # Linux 启动脚本
└── Stear.bat               # Windows 启动脚本
